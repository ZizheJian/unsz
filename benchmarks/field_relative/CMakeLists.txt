cmake_minimum_required(VERSION 3.19)
project(field_relative_baselines LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG -march=native")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SZ3_BINARY OFF CACHE BOOL "" FORCE)
set(BUILD_UTILITIES OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../third_party/SZ3 sz3)
find_package(OpenMP REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
add_executable(compress_one_field_with_sz3_partitions compress_one_field_with_partitions.cpp)
target_compile_definitions(compress_one_field_with_sz3_partitions PRIVATE FIELD_RELATIVE_SZ3=1)
target_link_libraries(compress_one_field_with_sz3_partitions PRIVATE SZ3::SZ3 OpenMP::OpenMP_CXX)
add_executable(compress_one_field_with_qoz_partitions compress_one_field_with_partitions.cpp)
target_compile_definitions(compress_one_field_with_qoz_partitions PRIVATE FIELD_RELATIVE_QOZ=1 QoZ_DEBUG_TIMINGS=0)
target_include_directories(compress_one_field_with_qoz_partitions PRIVATE ../../third_party/QoZ/include)
target_link_libraries(compress_one_field_with_qoz_partitions PRIVATE OpenMP::OpenMP_CXX PkgConfig::ZSTD)
