cmake_minimum_required(VERSION 3.19)
project(tensor_nd_baseline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG -march=native")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SZ3_BINARY OFF CACHE BOOL "" FORCE)
set(BUILD_UTILITIES OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../third_party/SZ3 sz3)
find_package(OpenMP REQUIRED)
add_executable(benchmark_sz3 benchmark.cpp)
target_link_libraries(benchmark_sz3 PRIVATE SZ3::SZ3 OpenMP::OpenMP_CXX)
