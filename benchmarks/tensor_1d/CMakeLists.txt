cmake_minimum_required(VERSION 3.19)
project(tensor_1d_baselines LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG -march=native")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SZ3_BINARY OFF CACHE BOOL "" FORCE)
set(BUILD_UTILITIES OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZFP_WITH_OPENMP ON CACHE BOOL "" FORCE)
add_subdirectory(../../third_party/SZ3 sz3)
add_subdirectory(../../third_party/zfp zfp)
find_package(OpenMP REQUIRED)
foreach(codec sz3 zfp pfpl)
    add_executable(benchmark_${codec} benchmark.cpp)
    target_compile_definitions(benchmark_${codec} PRIVATE BASELINE_${codec}=1)
    target_link_libraries(benchmark_${codec} PRIVATE OpenMP::OpenMP_CXX)
endforeach()
target_link_libraries(benchmark_sz3 PRIVATE SZ3::SZ3)
target_link_libraries(benchmark_zfp PRIVATE zfp)
target_sources(benchmark_pfpl PRIVATE pfpl_compress.cpp pfpl_decompress.cpp)
target_compile_options(benchmark_pfpl PRIVATE -mno-fma -ffp-contract=off)
